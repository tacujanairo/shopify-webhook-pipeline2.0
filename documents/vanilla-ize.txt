const http = require('http');

http.createServer((req, res) => {
    const chunks = [];

    req.on('data', chunk => {
        chunks.push(chunk);
    });

    req.on('end', () => {
        // This is basically what Express gives you as `buf`
        const buf = Buffer.concat(chunks);

        // Equivalent to:
        // req.rawBody = buf;
        req.rawBody = buf;

        // Equivalent to Express parsing JSON
        req.body = JSON.parse(buf.toString('utf8'));

        console.log(req.rawBody);
        console.log(req.body);

        res.end('OK');
    });
}).listen(3000);
